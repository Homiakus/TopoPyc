@topojit(opt_level="-O3")
def kernel_mandelbrot(c_re: float, c_im: float, max_iter: int) -> int:
    z_re = 0.0
    z_im = 0.0
    for i in range(max_iter):
        z_re2 = z_re * z_re
        z_im2 = z_im * z_im
        if (z_re2 + z_im2) > 4.0:
            return i
        z_im = 2.0 * z_re * z_im + c_im
        z_re = z_re2 - z_im2 + c_re
    return max_iter

In-Python JIT Kernel Python API 
from topopyc import topojit
import math

@topojit(opt_level="-O3")
def heavy_kernel(x: float, steps: int) -> float:
    acc = x
    for i in range(steps):
        acc += math.sin(acc * 0.01)
    return acc

# First call triggers JIT compilation (<30ms)
# All subsequent calls run at bare-metal speed
result = heavy_kernel(1.0, 10_000_000)
